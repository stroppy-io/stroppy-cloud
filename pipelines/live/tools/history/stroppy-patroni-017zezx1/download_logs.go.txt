  var logFile *os.File
  if strings.HasSuffix(b.Ref,"-log") { logFile,err=os.OpenFile(filepath.Join(os.Args[2],filepath.Base(b.Ref)+".log"),os.O_CREATE|os.O_TRUNC|os.O_WRONLY,0600);if err!=nil{panic(err)} }
  for {msg,err:=stream.Recv();if err==io.EOF{break};if err!=nil{panic(err)};chunk:=msg.GetChunk();hash.Write(chunk);if logFile!=nil {if _,err=logFile.Write(chunk);err!=nil {panic(err)}};size+=int64(len(chunk))}
  if logFile!=nil {if err=logFile.Close();err!=nil{panic(err)}}
  cancel();digest:="sha256:"+hex.EncodeToString(hash.Sum(nil));if digest!=b.Digest||size!=b.Size{panic("blob checksum mismatch: "+b.Ref)};b.Verified=true
 }
 raw,err=json.MarshalIndent(blobs,"","  ");if err!=nil{panic(err)};fmt.Println(string(raw))
}
